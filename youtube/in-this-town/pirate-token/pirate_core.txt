const fly = (p0, v, t, g = -22, floor = 0.4, rest = 0.35) => { let p = p0.clone(), vel = v.clone(), tt = t; for (let b = 0; b < 4; b++) { const a = 0.5 * g, bq = vel.y, cq = p.y - floor; const disc = bq * bq - 4 * a * cq; const th = disc > 0 ? (-bq - Math.sqrt(disc)) / (2 * a) : 1e9;
  if (tt < th || th <= 0) { return V(p.x + vel.x * tt, Math.max(floor, p.y + vel.y * tt + a * tt * tt), p.z + vel.z * tt); } p = V(p.x + vel.x * th, floor, p.z + vel.z * th); vel = V(vel.x * 0.55, -(vel.y + g * th) * rest, vel.z * 0.55); tt -= th; if (Math.abs(vel.y) < 1) { const slide = Math.min(tt, 0.4); return V(p.x + vel.x * slide, floor, p.z + vel.z * slide); } } return p; };
